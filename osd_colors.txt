# MEGA-SNES Pi Metal - OSD Color Overrides
#
# Optional: override specific OSD colors after the base theme is loaded.
# Place this file at SD:/osd_colors.txt (or SD:/roms/osd_colors.txt).
#
# Format:
#   key = value
#
# Value formats:
#   - RGB555 triplet: r,g,b     (0..31 each)
#   - RGB888 triplet: r,g,b     (0..255 each)
#   - Hex RGB888: #RRGGBB
#   - Hex RGB555: 0x7FFF
#
# Common keys:
#   background, panel_background, border, text, warning_text
#
# Advanced keys:
#   screen_bg, panel_bg, panel_border, title_text, separator
#   tab_active_bg, tab_active_border, tab_active_text
#   tab_inactive_bg, tab_inactive_border, tab_inactive_text
#   list_selected_text, list_text, list_selected_bg
#   scroll_track, scroll_thumb, footer_text
#
# Example overrides (uncomment to use):
# background = 0,0,0
# panel_background = #101820
# border = 8,12,16
# text = 26,28,30
#
